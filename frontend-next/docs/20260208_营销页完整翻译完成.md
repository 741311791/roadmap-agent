### 组件翻译模式

```typescript
import { useTranslations } from 'next-intl';

export function MyComponent() {
  const t = useTranslations('sectionName');
  
  return <h1>{t('title')}</h1>;
}
```

### 动态列表翻译

```typescript
const t = useTranslations('agents');

const agents = [
  { titleKey: 'intentAnalyzer', descriptionKey: 'intentAnalyzerDesc', ... },
  // ...
];

return (
  <div>
    {agents.map(agent => (
      <div key={agent.titleKey}>
        <h3>{t(agent.titleKey as any)}</h3>
        <p>{t(agent.descriptionKey as any)}</p>
      </div>
    ))}
  </div>
);
```
