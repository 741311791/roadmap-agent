```python
# app/utils/serializers.py (新建)
"""
高性能序列化工具（基于msgspec）
"""
import msgspec
from typing import Any, TypeVar, Type

T = TypeVar('T')

# ===== 全局编码器/解码器 =====

_encoder = msgspec.json.Encoder()
_decoder = msgspec.json.Decoder()

def fast_dumps(obj: Any) -> bytes:
    """
    高性能JSON序列化
    
    性能：比json.dumps()快5-10倍
    支持：Pydantic模型、dataclass、普通dict
    """
    return _encoder.encode(obj)

def fast_loads(data: bytes | str, type_: Type[T] | None = None) -> T | Any:
    """
    高性能JSON反序列化
    
    Args:
        data: JSON字符串或字节
        type_: 目标类型（可选，用于类型安全）
        
    Returns:
        反序列化后的对象
    """
    if isinstance(data, str):
        data = data.encode()
    
    if type_:
        return msgspec.json.decode(data, type=type_)
    return _decoder.decode(data)

# ===== 类型化编码器（支持Pydantic）=====

def typed_encoder(type_: Type[T]) -> msgspec.json.Encoder:
    """
    创建类型化编码器
    
    用于Pydantic模型的高性能序列化
    """
    return msgspec.json.Encoder()

def typed_decoder(type_: Type[T]) -> msgspec.json.Decoder:
    """
    创建类型化解码器
    
    用于Pydantic模型的高性能反序列化
    """
    return msgspec.json.Decoder(type=type_)
```

### 任务2: 修复LangGraph Checkpointer阻塞 [Day 3-4, 10h]

#### 方案A: ThreadPoolExecutor包装（推荐）

```python
# app/core/orchestrator_factory.py (修改)
"""
Orchestrator工厂 - 解决Checkpointer同步阻塞问题
"""
import asyncio
from concurrent.futures import ThreadPoolExecutor
from psycopg_pool import ConnectionPool
from langgraph_checkpoint_postgres import PostgresSaver
import structlog

from app.config.settings import settings

logger = structlog.get_logger()

class AsyncCheckpointerWrapper:
    """
    Checkpointer异步包装器
    
    原理：
    - 将同步的PostgresSaver操作放到独立线程池执行
    - 避免阻塞主事件循环
    - 保持与LangGraph的兼容性
    """
    
    def __init__(self, sync_checkpointer: PostgresSaver):
        self._checkpointer = sync_checkpointer
        # ✅ 专用线程池（4个线程足够）
        self._executor = ThreadPoolExecutor(
            max_workers=4,
            thread_name_prefix="checkpointer_"
        )
        logger.info("async_checkpointer_wrapper_initialized")
    
    async def put(self, config: dict, checkpoint: dict, metadata: dict):
        """
        异步保存Checkpoint
        
        Args:
            config: 配置（包含thread_id）
            checkpoint: Checkpoint数据
            metadata: 元数据
        """
        loop = asyncio.get_event_loop()
        # ✅ 在独立线程执行同步操作
        await loop.run_in_executor(
            self._executor,
            self._checkpointer.put,
            config,
            checkpoint,
            metadata
        )
        logger.debug(
            "checkpoint_saved_async",
            thread_id=config.get("thread_id"),
        )
    
    async def get(self, config: dict):
        """
        异步加载Checkpoint
        
        Args:
            config: 配置（包含thread_id）
            
        Returns:
            Checkpoint数据
        """
        loop = asyncio.get_event_loop()
        return await loop.run_in_executor(
            self._executor,
            self._checkpointer.get,
            config
        )
    
    async def list(self, config: dict):
        """异步列出Checkpoints"""
        loop = asyncio.get_event_loop()
        return await loop.run_in_executor(
            self._executor,
            self._checkpointer.list,
            config
        )
    
    def __del__(self):
        """清理线程池"""
        self._executor.shutdown(wait=True)
        logger.info("async_checkpointer_wrapper_destroyed")

def create_checkpointer() -> AsyncCheckpointerWrapper:
    """
    创建异步Checkpointer
    
    Returns:
        AsyncCheckpointerWrapper实例
    """
    # ✅ 创建同步连接池（LangGraph要求）
    pool = ConnectionPool(
        conninfo=settings.CHECKPOINTER_DATABASE_URL,
        min_size=2,
        max_size=10,
        timeout=30,
    )
    
    sync_checkpointer = PostgresSaver(pool)
    
    # ✅ 包装为异步接口
    async_checkpointer = AsyncCheckpointerWrapper(sync_checkpointer)
    
    logger.info(
        "checkpointer_created",
        pool_size=f"{pool.min_size}-{pool.max_size}",
    )
    
    return async_checkpointer
```

```python
from sqlalchemy.orm import selectinload, joinedload

class RoadmapCRUD(BaseCRUD[RoadmapMetadata, ...]):
    async def get_with_concepts(
        self,
        session: AsyncSession,
        roadmap_id: str,
    ) -> Optional[RoadmapMetadata]:
        """
        获取路线图及其所有概念元数据（避免N+1查询）
        
        使用selectinload预加载关联数据
        """
        result = await session.execute(
            select(RoadmapMetadata)
            .options(
                # ✅ 预加载concepts
                selectinload(RoadmapMetadata.concept_metas),
            )
            .where(RoadmapMetadata.roadmap_id == roadmap_id)
        )
        return result.scalar_one_or_none()
    
    async def get_with_all_relations(
        self,
        session: AsyncSession,
        roadmap_id: str,
    ) -> Optional[RoadmapMetadata]:
        """
        获取路线图及其所有关联数据（避免N+1查询）
        
        使用多级selectinload
        """
        result = await session.execute(
            select(RoadmapMetadata)
            .options(
                # ✅ 预加载concepts
                selectinload(RoadmapMetadata.concept_metas).selectinload(
                    # ✅ 预加载tutorial_metas
                    ConceptMetadata.tutorial_metas
                ),
                selectinload(RoadmapMetadata.concept_metas).selectinload(
                    # ✅ 预加载resource_metas
                    ConceptMetadata.resource_metas
                ),
                selectinload(RoadmapMetadata.concept_metas).selectinload(
                    # ✅ 预加载quiz_metas
                    ConceptMetadata.quiz_metas
                ),
            )
            .where(RoadmapMetadata.roadmap_id == roadmap_id)
        )
        return result.scalar_one_or_none()
```

```python
# app/middleware/trace_middleware.py (新建)
"""
TraceID中间件 - 请求链路追踪
"""
import uuid
from starlette.middleware.base import BaseHTTPMiddleware
from starlette.requests import Request
import structlog

logger = structlog.get_logger()

class TraceIDMiddleware(BaseHTTPMiddleware):
    """
    为每个请求注入trace_id，支持分布式追踪
    
    功能：
    1. 从请求头读取X-Trace-ID（如果有）
    2. 否则生成新的UUID
    3. 注入到request.state（后续可在日志中访问）
    4. 在响应头中返回X-Trace-ID
    5. 绑定到structlog上下文（所有日志自动包含trace_id）
    """
    
    async def dispatch(self, request: Request, call_next):
        # ✅ 优先使用客户端传入的trace_id（跨服务追踪）
        trace_id = request.headers.get("X-Trace-ID") or str(uuid.uuid4())
        
        # ✅ 注入到request.state
        request.state.trace_id = trace_id
        
        # ✅ 绑定到structlog上下文
        with structlog.contextvars.bound_contextvars(trace_id=trace_id):
            logger.info(
                "request_started",
                method=request.method,
                path=request.url.path,
                client_ip=request.client.host,
            )
            
            try:
                # 调用下游中间件/路由
                response = await call_next(request)
                
                # ✅ 在响应头中返回trace_id
                response.headers["X-Trace-ID"] = trace_id
                
                logger.info(
                    "request_completed",
                    status_code=response.status_code,
                )
                
                return response
                
            except Exception as e:
                logger.error(
                    "request_failed",
                    error=str(e),
                    exc_info=True,
                )
                raise
```

```python
状态管理器（Redis版本）
"""
from app.db.redis_client import redis_client
import structlog

logger = structlog.get_logger()

class StateManager:
    """
    状态管理器（Redis版本）
    
    优势：
    - 支持水平扩展（多实例共享状态）
    - 持久化（重启不丢失）

    LIVE_STEP_PREFIX = "workflow:live_step:"
    LIVE_STEP_TTL = 3600  # 1小时
    
    async def set_live_step(self, task_id: str, step: str):
        """
        设置任务当前步骤（Redis）
        
        Args:
            task_id: 任务ID
            step: 步骤名称
        """
        await redis_client.connect()
        key = f"{self.LIVE_STEP_PREFIX}{task_id}"
        
        await redis_client._client.setex(
            key,
            self.LIVE_STEP_TTL,
            step,
        )
        
        logger.debug(
            "live_step_set",
            task_id=task_id,
            step=step,
        )
    
    async def get_live_step(self, task_id: str) -> str | None:
        """
        获取任务当前步骤
        
        Args:
            task_id: 任务ID
            
        Returns:
            步骤名称或None
        """
        await redis_client.connect()
        key = f"{self.LIVE_STEP_PREFIX}{task_id}"
        
        step = await redis_client._client.get(key)
        
        if step:
            logger.debug(
                "live_step_retrieved",
                task_id=task_id,
                step=step,
            )
        
        return step
    
    async def clear_live_step(self, task_id: str):
        """
        清除任务步骤缓存
        
        Args:
            task_id: 任务ID
        """
        await redis_client.connect()
        key = f"{self.LIVE_STEP_PREFIX}{task_id}"
        
        await redis_client._client.delete(key)
        
        logger.debug(
            "live_step_cleared",
            task_id=task_id,
        )
```

#### 连接池计算

```python
# app/config/settings.py (修改)

class Settings(BaseSettings):
    # ===== 主数据库连接池（每进程）=====
    DB_POOL_SIZE: int = Field(4, description="主数据库连接池大小")
    DB_MAX_OVERFLOW: int = Field(3, description="主数据库最大溢出")
    
    # ===== LangGraph Checkpointer连接池（全局共享）=====
    LANGGRAPH_POOL_MIN: int = Field(2, description="LangGraph最小连接数")
    LANGGRAPH_POOL_MAX: int = Field(10, description="LangGraph最大连接数")
    
    @property
    def estimated_total_connections(self) -> int:
        """
        估算总连接数
        
        公式：主池 × 进程数 + LangGraph池
        
        部署配置：
        - API Workers: 4个uvicorn进程
        - Celery Workers: 3个worker进程
        - 总计：7个进程
        """
        api_workers = 4
        celery_workers = 3
        total_workers = api_workers + celery_workers
        
        main_pool = (self.DB_POOL_SIZE + self.DB_MAX_OVERFLOW) * total_workers
        langgraph_pool = self.LANGGRAPH_POOL_MAX
        
        total = main_pool + langgraph_pool
        
        logger.info(
            "connection_pool_calculation",
            api_workers=api_workers,
            celery_workers=celery_workers,
            main_pool_per_worker=self.DB_POOL_SIZE + self.DB_MAX_OVERFLOW,
            main_pool_total=main_pool,
            langgraph_pool=langgraph_pool,
            estimated_total=total,
        )
        
        return total

settings = Settings()

# 启动时检查
if settings.estimated_total_connections > 90:
    logger.warning(
        "connection_pool_may_exceed_limit",
        estimated=settings.estimated_total_connections,
        postgres_limit=100,
        recommendation="Reduce pool size or worker count",
    )
```
