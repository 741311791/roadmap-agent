    async def update_concept_progress(self, session, user_id, roadmap_id, concept_id, is_completed):
        return await self.progress_crud.upsert_concept_progress(...)

@router.put("/roadmaps/{roadmap_id}/concepts/{concept_id}")
async def update_concept_progress(
    roadmap_id, concept_id, payload, session,
