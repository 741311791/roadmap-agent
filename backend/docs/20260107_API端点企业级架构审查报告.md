## 🔴 严重违规问题（Critical）

### 1. admin.py - Session 管理混乱

**问题描述**:  
使用 `get_db_transaction` 但手动调用 `commit()/rollback()`，违反企业级架构规范。

**当前代码** (第52-92行):
```python
@router.post("/invite-user", response_model=InviteUserResponse)
async def invite_user(
    request: InviteUserRequest,
    current_user: User = Depends(current_superuser),
    user_manager: UserManager = Depends(get_user_manager),
    email_service: EmailService = Depends(get_email_service),
    db: AsyncSession = Depends(get_db_transaction),  # ❌ 错误
):
    try:
        service = UserInviteService()
        result = await service.invite_single_user(...)
        
        await db.commit()  # ❌ 不应手动commit
        
        return InviteUserResponse(**result)
        
    except ValueError as e:
        await db.rollback()  # ❌ 不应手动rollback
        raise HTTPException(status_code=400, detail=str(e))
```

**规范要求** (FastAPI_Enterprise_Architecture_Guide.md 第262-283行):
```python
# ✅ 写操作 Session (自动commit/rollback)
async def get_db_transaction() -> AsyncGenerator[AsyncSession, None]:
    """写操作使用"""
    async with async_db_session.begin() as session:
        yield session
        # 成功自动commit，异常自动rollback

CurrentSessionTransaction = Annotated[AsyncSession, Depends(get_db_transaction)]
```

**修复方案**:
```python
from app.api.v1.deps import CurrentSessionTransaction

@router.post("/invite-user", response_model=InviteUserResponse)
async def invite_user(
    request: InviteUserRequest,
    db: CurrentSessionTransaction,  # ✅ 自动管理事务
    current_user: User = Depends(current_superuser),
    user_manager: UserManager = Depends(get_user_manager),
    email_service: EmailService = Depends(get_email_service),
):
    try:
        service = UserInviteService()
        result = await service.invite_single_user(...)
        return InviteUserResponse(**result)
        # ✅ 成功自动commit，异常自动rollback
        
    except ValueError as e:
        raise HTTPException(status_code=400, detail=str(e))
```

**影响范围**:
- `admin.py` 全部9个端点
- `edit.py`, `validation.py` 类似问题

### 2. 缺少统一响应格式

**问题描述**:  
大部分API直接返回字典，未使用 `ResponseModel/ResponseSchemaModel`，导致响应格式不统一。

**违规文件**:
- `content.py` (全部9个端点)
- `retrieval.py`
- `status.py`
- `management.py`

**当前代码** (content.py 第50-67行):
```python
@router.get("/{roadmap_id}/concepts/{concept_id}/tutorials")
async def get_tutorial_versions(...):
    tutorials = await service.get_tutorial_versions(...)
    
    # ❌ 直接返回字典
    return {
        "roadmap_id": roadmap_id,
        "concept_id": concept_id,
        "total_versions": len(tutorials),
        "tutorials": [...]
    }
```

**规范要求** (FastAPI_Enterprise_Architecture_Guide.md 第419-450行):
```python
class ResponseSchemaModel(ResponseModel, Generic[SchemaT]):
    """泛型响应模型（带类型提示）"""
    data: SchemaT

@router.get("/users/{pk}")
async def get_user(pk: int) -> ResponseSchemaModel[GetUserInfoDetail]:
    user = await user_service.get_userinfo(pk=pk)
    return response_base.success(data=user)
```

**修复方案**:
```python
# 1. 定义响应Schema (app/schemas/content.py)
class TutorialVersionListResponse(SchemaBase):
    roadmap_id: str
    concept_id: str
    total_versions: int
    tutorials: list[TutorialItemResponse]

# 2. 使用统一响应格式
from app.common.response.response_code import response_base
from app.common.response.response_schema import ResponseSchemaModel

@router.get("/{roadmap_id}/concepts/{concept_id}/tutorials")
async def get_tutorial_versions(
    ...
) -> ResponseSchemaModel[TutorialVersionListResponse]:
    tutorials = await service.get_tutorial_versions(...)
    
    if not tutorials:
        raise errors.NotFoundError(msg=f"概念 {concept_id} 没有教程")
    
    # ✅ 使用统一响应封装
    return response_base.success(data=TutorialVersionListResponse(
        roadmap_id=roadmap_id,
        concept_id=concept_id,
        total_versions=len(tutorials),
        tutorials=tutorials
    ))
```

```python
# backend/app/admin/schema/user.py
```

**问题描述**:  
`generate_roadmap_async` 端点包含复杂的任务验证逻辑（第76-138行），应该移到Service层。

```python
@router.post("/generate", response_model=GenerateRoadmapResponse)
async def generate_roadmap_async(
    request: UserRequest,
    repo_factory: CurrentRepoFactory,
):
    # ❌ API层包含大量业务逻辑
    task_id = str(uuid.uuid4())
    
    # 创建任务
    async with repo_factory.create_session() as session:
        task_repo = repo_factory.create_task_repo(session)
        await task_repo.create_task(...)
        await session.commit()
    
    # 验证任务持久化（50行复杂逻辑）
    max_verify_retries = 5
    task_verified = False
    
    for attempt in range(max_verify_retries):
        async with repo_factory.create_session() as session:
            ...
        if attempt < max_verify_retries - 1:
            await asyncio.sleep(0.05 * (attempt + 1))
    
    # 分发Celery任务
    celery_task = generate_roadmap.delay(...)
    
    # 更新任务记录
    async with repo_factory.create_session() as session:
        ...
```

**修复方案**:
```python
# 1. 创建 GenerationService
# app/services/generation_service.py
class GenerationService:
    async def create_and_verify_task(
        self, 
        repo_factory: RepositoryFactory,
        user_request: UserRequest
    ) -> tuple[str, str]:
        """创建任务并验证持久化，返回 (task_id, celery_task_id)"""
        # 所有业务逻辑移到这里
        ...

# 2. API层瘦身
@router.post("/generate", response_model=GenerateRoadmapResponse)
async def generate_roadmap_async(
    request: UserRequest,
    repo_factory: CurrentRepoFactory,
    service: GenerationService = Depends(),
) -> ResponseSchemaModel[GenerateRoadmapResponse]:
    """生成学习路线图（API层只负责HTTP适配）"""
    try:
        task_id, celery_task_id = await service.create_and_verify_task(
            repo_factory, request
```

## 🔧 统一修复模板

### 模板 1: Session 管理标准化
```python
# ❌ 旧代码
@router.post("/xxx")
async def create_xxx(
    db: AsyncSession = Depends(get_db_transaction),
):
    try:
        result = await service.create(db, ...)
        await db.commit()
        return result
    except Exception as e:
        await db.rollback()
        raise HTTPException(status_code=500, detail=str(e))

# ✅ 新代码
from app.api.v1.deps import CurrentSessionTransaction
from app.common.response.response_code import response_base
from app.common.exception import errors

@router.post("/xxx", response_model=ResponseSchemaModel[XxxResponse])
async def create_xxx(
    db: CurrentSessionTransaction,
    obj: CreateXxxRequest,
) -> ResponseSchemaModel[XxxResponse]:
    """创建XXX（自动管理事务）"""
    try:
        result = await service.create(db, obj)
        return response_base.success(data=result)
    except ValueError as e:
        raise errors.RequestError(msg=str(e))
```
