```python
async def generate_tutorial_for_concept(state: ContentGenState) -> dict:
    """
    为单个 Concept 生成教程（集成 WorkflowBrain）
    """
    concept = state["concept"]
    brain = state["brain"]  # ✅ 使用传入的 brain
    agent_factory = state["agent_factory"]  # ✅ 使用传入的 factory
    
    # ✅ 使用 WorkflowBrain 的上下文管理器

        "task_id": state["task_id"],
        "roadmap_id": state["roadmap_id"],
        "current_step": f"generate_tutorial_{concept.concept_id}",

        # 创建Agent
        tutorial_agent = agent_factory.create_tutorial_generator()
        
        # 构造输入
        tutorial_input = TutorialGenerationInput(
            concept=concept,
            context=state.get("context", {}),
            user_preferences=state["user_preferences"],

            logger.info(
                "tutorial_generation_completed",
                task_id=state["task_id"],
                concept_id=concept.concept_id,
```

### 任务 1.3：配置 Node 级 RetryPolicy

```python
# backend/app/core/orchestrator/retry_policies.py
from langgraph.types import RetryPolicy
import litellm
from requests.exceptions import Timeout, ConnectionError

# ===== LLM调用的重试策略 =====
LLM_RETRY_POLICY = RetryPolicy(
    max_attempts=5,
    retry_on=(
        litellm.RateLimitError,  # 限流错误
        litellm.APIError,  # API错误
        litellm.Timeout,  # 超时
    ),
    backoff_factor=2.0,  # 指数退避：2s, 4s, 8s, 16s, 32s
    initial_interval=2.0,  # 初始间隔2秒
)

# ===== Tavily搜索的重试策略 =====
TAVILY_RETRY_POLICY = RetryPolicy(
    max_attempts=3,
    retry_on=(
        Timeout,  # 超时
        ConnectionError,  # 连接错误
        Exception,  # 通用异常（Tavily可能抛出各种错误）
    ),
    backoff_factor=1.5,  # 指数退避：1.5s, 2.25s, 3.375s
    initial_interval=1.5,
)

# ===== 纯逻辑节点（不重试） =====
NO_RETRY_POLICY = RetryPolicy(
    max_attempts=1,  # 不重试
)
```
