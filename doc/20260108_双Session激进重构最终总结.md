```python
# session.py
from app.db.session import safe_session_with_retry

async with safe_session_with_retry(max_retries=3, base_backoff=0.5) as session:
    result = await session.execute(query)
    await session.commit()  # 手动 commit

repo_factory = CeleryRepositoryFactory()

    await session.commit()
```

#### 新代码（简洁）

```python
# session.py
from app.db.session import async_session_maker

async with async_session_maker.begin() as session:
    result = await session.execute(query)
    # ✅ 自动 commit/rollback

# celery_session.py
from app.db.celery_session import get_celery_session
from app.crud.crud_task import get_task_crud

async with get_celery_session() as session:
    task_crud = get_task_crud()
    result = await task_crud.get(...)
    # ✅ 自动 commit/rollback
```
